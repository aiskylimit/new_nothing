# 6. INFERENCE PILOT: THU LOAD CHECKPOINT, SINH VA CHAM TREN 3 GPU
# Smoke o muc 5 KHONG tu eval. Pilot ben duoi dung checkpoint smoke cua moi run.
# Moi run chi lay 3 cau MATH500; A/C them 3 ARC + 3 LogiQA. Moi GPU nhan 1 cau/bench.

import json
import os
import subprocess
from pathlib import Path

from rebuttal.common import jsonl, matrix, rows, sha, write

repo = Path.cwd()
smokes = Path(os.environ['REBUTTAL_SMOKE_RUNS'])
benchmarks = Path(os.environ['REBUTTAL_DATA']) / 'benchmarks'
gpu_ids = os.environ['GPUS'].split(',')
if len(gpu_ids) != 3 or len(set(gpu_ids)) != 3:
    raise ValueError('Pilot requires three distinct GPUs')
evaluator = os.environ['REBUTTAL_PYTHON_PALIGN']
known = {c['run_id'] for c in matrix()}
selected = os.environ.get('PILOT_RUN_IDS', '').split() or [c['run_id'] for c in matrix()]
if set(selected) - known:
    raise ValueError('Unknown PILOT_RUN_IDS')
for run_id in selected:
    run = smokes / ('smoke_' + run_id)
    config = json.loads((run / 'run.json').read_text())
    checkpoint = Path(json.loads((run / 'final_checkpoint.json').read_text())['path'])
    pilot = run / 'inference-pilot'
    local_benchmarks = pilot / 'benchmarks'
    output = pilot / 'evaluation'
    output.mkdir(parents=True, exist_ok=True)
    config['run_id'] += '_inference_pilot'
    config['evaluation']['math'] = ['math500']
    config['evaluation']['ood'] = ['arc_challenge', 'logiqa2'] if config['setting'] != 'B' else []
    original_manifest = json.loads((benchmarks / 'manifest.json').read_text())
    manifest = {}
    for name in config['evaluation']['math'] + config['evaluation']['ood']:
        values = rows(benchmarks / (name + '.jsonl'))[:3]
        if len(values) != 3:
            raise ValueError(f'Need three pilot questions: {name}')
        destination = local_benchmarks / (name + '.jsonl')
        jsonl(destination, values)
        manifest[name] = dict(original_manifest[name], count=3, sha256=sha(destination), pilot=True)
    write(local_benchmarks / 'manifest.json', manifest)
    config_file = pilot / 'config.json'
    write(config_file, config)
    common = [evaluator, '-m', 'rebuttal.evaluate', '--config', str(config_file),
              '--benchmarks', str(local_benchmarks), '--output', str(output)]
    processes, logs = [], []
    print(f'[pilot] {run_id}: starting three GPU workers; logs in {pilot}', flush=True)
    try:
        for rank, gpu in enumerate(gpu_ids):
            log = (pilot / f'generate-rank{rank}.log').open('a')
            logs.append(log)
            command = common + ['--checkpoint', str(checkpoint), '--shard', str(rank),
                                '--shards', '3', '--batch-size', '1']
            processes.append(subprocess.Popen(command, cwd=repo,
                env={**os.environ, 'CUDA_VISIBLE_DEVICES': gpu}, stdout=log, stderr=subprocess.STDOUT))
        for rank, process in enumerate(processes):
            if process.wait() != 0:
                raise RuntimeError(f'Pilot worker {rank} failed; see {pilot}/generate-rank{rank}.log')
    finally:
        for process in processes:
            if process.poll() is None:
                process.terminate()
                try:
                    process.wait(timeout=15)
                except subprocess.TimeoutExpired:
                    process.kill()
                    process.wait()
        for log in logs:
            log.close()
    subprocess.run(common + ['--score-only'], cwd=repo,
                   env={**os.environ, 'CUDA_VISIBLE_DEVICES': ''}, check=True)
    print(f'[pilot] PASS {run_id}: generated and scored complete pilot records', flush=True)
print('ALL SELECTED INFERENCE PILOTS PASSED; this is not full benchmark accuracy.', flush=True)
PILOT_PY
