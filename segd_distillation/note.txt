# ---------------------------------------------------------------------------
# 0. Go to repo root
# ---------------------------------------------------------------------------
cd segd_distillation

# ---------------------------------------------------------------------------
# 1. Setup env
# ---------------------------------------------------------------------------
apt-get update
apt-get upgrade -y
python -m venv vlm
source vlm/bin/activate
pip install -r requirements.txt
python fix_lib.py

# ---------------------------------------------------------------------------
# 2. Download train images (MMEB-train, cls subsets only)
#    ImageNet_1K, N24News, HatefulMemes, VOC2007, SUN397
# ---------------------------------------------------------------------------
pip install -U huggingface_hub hf_transfer
export HF_HUB_ENABLE_HF_TRANSFER=1
python download.py

# ---------------------------------------------------------------------------
# 3. Download eval images (MMEB-eval)
# ---------------------------------------------------------------------------
hf download TIGER-Lab/MMEB-eval images.zip --repo-type dataset --local-dir .
mkdir -p eval_images
unzip -o images.zip -d eval_images/
rm -f images.zip
# ImageNet-1K only (optional):
# unzip images.zip "ImageNet-1K/*" -d eval_images/

# ---------------------------------------------------------------------------
# 4. Train SEGD (classification metatask)
#    Edit hyperparams in scripts/cls/run_segd_cls.sh (do not edit the interface).
# ---------------------------------------------------------------------------
bash scripts/cls/run_segd_cls.sh

# Or override without editing:
# EXP_NAME=my_segd_run LEARNING_RATE=2e-4 SEGD_LAMBDA_SPECTRAL=0.5 bash scripts/cls/run_segd_cls.sh
# bash scripts/cls/run_segd_cls.sh --learning_rate 2e-4

# Checkpoints written to:
#   training/FastVLM-0.5B_segd_eos_cls/checkpoint-epoch-0
#   training/FastVLM-0.5B_segd_eos_cls/checkpoint-final

# Useful SEGD flags (already set in the script):
#   --kd_loss_type segd_loss
#   --segd_lambda_sim 1.0
#   --segd_lambda_spectral 1.0
#   --segd_tau_graph 1.0
#   --segd_num_align_layers 4
#   --segd_k_eigen_min 8
#   --num_projectors 1          # required for FastVLM 896 → Qwen2 1536 (L_sim)
#   --pooling eos
#   --teacher_pooling eos

# ---------------------------------------------------------------------------
# 5. Eval on MMEB (classification subsets)
# ---------------------------------------------------------------------------
bash scripts/cls/eval_segd_cls.sh

# Override checkpoint / exp name:
# EXP_NAME=my_segd_run MODEL=training/my_segd_run/checkpoint-final bash scripts/cls/eval_segd_cls.sh
