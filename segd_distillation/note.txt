#!/usr/bin/env bash
# SEGD classification command notes

export PROJECT_ROOT="/mnt/local/aiskylimit_new_nothing/segd_distillation"
export HF_HOME="${PROJECT_ROOT}/.hf_home"
cd "$PROJECT_ROOT"

# ---------------------------------------------------------------------------
# 1. Setup once
# ---------------------------------------------------------------------------
python3 -m venv .venv
source .venv/bin/activate
python -m pip install -r requirements.txt

# ---------------------------------------------------------------------------
# 2. Download once; skip this section when data/models already exist
# ---------------------------------------------------------------------------
while read -r mode source destination; do
    [[ -z "${mode:-}" || "$mode" == \#* ]] && continue
    mkdir -p "$destination"
    case "$mode" in
        --url) wget --continue --directory-prefix="$destination" "$source" ;;
        --hf)  hf download "$source" --local-dir "$destination" ;;
    esac
done < download.txt

# B3_Qwen2_2B is an adapter; cache its base model too.
hf download Qwen/Qwen2-VL-2B-Instruct

# ---------------------------------------------------------------------------
# 3. Recommended: run the complete pipeline
# ---------------------------------------------------------------------------
bash project_commands.sh

# Outputs:
# training/FastVLM-0.5B_segd_eos_cls_full/checkpoint-final/
# MMEB-evaloutputs-json/FastVLM-0.5B_segd_eos_cls_full/

# ---------------------------------------------------------------------------
# 4. Run individual stages only when needed
# ---------------------------------------------------------------------------

# Train only:
CUDA_VISIBLE_DEVICES=1 \
MODEL_NAME="${PROJECT_ROOT}/models/FastVLM-0.5B" \
TEACHER_MODEL_NAME="${PROJECT_ROOT}/models/B3_Qwen2_2B" \
EXP_NAME="FastVLM-0.5B_segd_eos_cls_full" \
PERCENT_DATA=1.0 \
bash scripts/cls/run_segd_cls.sh

# Eval only:
CUDA_VISIBLE_DEVICES=1 \
EXP_NAME="FastVLM-0.5B_segd_eos_cls_full" \
MODEL="training/FastVLM-0.5B_segd_eos_cls_full/checkpoint-final" \
bash scripts/cls/eval_segd_cls.sh

# Keep JSON scores only:
python json_filter.py \
    ./MMEB-eval_outputs \
    ./MMEB-evaloutputs-json \
    --overwrite
